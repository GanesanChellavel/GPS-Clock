#include <Arduino.h>
#include <ESP32-HUB75-MatrixPanel-I2S-DMA.h>

// ---- Panel config ----
#define PANEL_WIDTH    64
#define PANEL_HEIGHT   32
#define PANELS_NUMBER  1

// ---- LoRa UART ----
#define RXD2 16
#define TXD2 17

// ---- LoRa Mode Pins ----
#define M0 18
#define M1 19

// ---- Button ----
#define BUTTON_PIN 21

MatrixPanel_I2S_DMA *dma_display = nullptr;
uint16_t currentColor;

// Default RED
uint16_t colorRed, colorGreen, colorBlue, colorYellow, colorWhite;

String receivedTime = "--:--";

// ---- Blink ----
bool showColon = true;
unsigned long lastBlink = 0;

// ---- Button variables ----
bool buttonState = HIGH;
bool lastButtonState = HIGH;
unsigned long lastDebounceTime = 0;
unsigned long debounceDelay = 50;

unsigned long pressStartTime = 0;
bool buttonPressed = false;

// ---- Extract TIME ----
String extractTime(String msg) {
  msg.trim();

  if (msg.startsWith("TIME:")) {
    String t = msg.substring(5);
    t.trim();

    if (t.length() == 5 && t.charAt(2) == ':') {
      return t;
    }
  }
  return "";
}

// ---- Display ----
void showTime(String timeStr) {
  dma_display->clearScreen();

  dma_display->setTextSize(2);
  dma_display->setTextColor(currentColor);
  dma_display->setTextWrap(false);

  String displayTime = timeStr;

  if (!showColon && displayTime.length() == 5) {
    displayTime.setCharAt(2, ' ');
  }

  int x = (PANEL_WIDTH - (5 * 12)) / 2;
  int y = (PANEL_HEIGHT - 16) / 2;

  dma_display->setCursor(x, y);
  dma_display->print(displayTime);
}

// ---- Handle Button ----
void handleButton() {
  int reading = digitalRead(BUTTON_PIN);

  // Debounce
  if (reading != lastButtonState) {
    lastDebounceTime = millis();
  }

  if ((millis() - lastDebounceTime) > debounceDelay) {
    if (reading != buttonState) {
      buttonState = reading;

      // Button PRESSED
      if (buttonState == LOW) {
        pressStartTime = millis();
        buttonPressed = true;
      }

      // Button RELEASED
      if (buttonState == HIGH && buttonPressed) {
        unsigned long pressDuration = millis() - pressStartTime;
        buttonPressed = false;

        Serial.print("Press Duration: ");
        Serial.println(pressDuration);

        // ---- Color Selection ----
        if (pressDuration >= 8000) {
          currentColor = colorWhite;
          Serial.println("Color: WHITE");
        }
        else if (pressDuration >= 6000) {
          currentColor = colorYellow;
          Serial.println("Color: YELLOW");
        }
        else if (pressDuration >= 4000) {
          currentColor = colorBlue;
          Serial.println("Color: BLUE");
        }
        else if (pressDuration >= 2000) {
          currentColor = colorGreen;
          Serial.println("Color: GREEN");
        }
        else {
          Serial.println("Too short press (ignored)");
        }
      }
    }
  }

  lastButtonState = reading;
}

void setup() {
  Serial.begin(115200);

  // ---- Button ----
  pinMode(BUTTON_PIN, INPUT_PULLUP);

  // ---- LoRa ----
  pinMode(M0, OUTPUT);
  pinMode(M1, OUTPUT);
  digitalWrite(M0, LOW);
  digitalWrite(M1, LOW);

  Serial2.begin(9600, SERIAL_8N1, RXD2, TXD2);

  // ---- HUB75 ----
  HUB75_I2S_CFG::i2s_pins _pins = {
    .r1 = 1, .g1 = 2, .b1 = 3,
    .r2 = 4, .g2 = 5, .b2 = 6,
    .a  = 7, .b  = 8, .c  = 9,
    .d  = 10, .e  = 11,
    .lat = 13, .oe  = 14, .clk = 12
  };

  HUB75_I2S_CFG mxconfig(PANEL_WIDTH, PANEL_HEIGHT, PANELS_NUMBER, _pins);
  mxconfig.clkphase = false;
  mxconfig.i2sspeed = HUB75_I2S_CFG::HZ_10M;

  dma_display = new MatrixPanel_I2S_DMA(mxconfig);
  dma_display->begin();
  dma_display->setBrightness8(100);
  dma_display->clearScreen();

  // ---- Colors ----
  colorRed    = dma_display->color565(255, 0, 0);
  colorGreen  = dma_display->color565(0, 255, 0);
  colorBlue   = dma_display->color565(0, 0, 255);
  colorYellow = dma_display->color565(255, 255, 0);
  colorWhite  = dma_display->color565(255, 255, 255);

  currentColor = colorRed;  // default

  dma_display->setTextSize(1);
  dma_display->setTextColor(colorRed);
  dma_display->setCursor(8, 12);
  dma_display->print("Waiting...");
}

void loop() {

  // ---- LoRa Receive ----
  if (Serial2.available()) {
    String msg = Serial2.readStringUntil('\n');
    msg.trim();

    Serial.println("Received: " + msg);

    String t = extractTime(msg);
    if (t != "") {
      receivedTime = t;
    }
  }

  // ---- Button ----
  handleButton();

  // ---- Blink ----
  if (millis() - lastBlink >= 1000) {
    showColon = !showColon;
    lastBlink = millis();
  }

  // ---- Display ----
  showTime(receivedTime);

  delay(50);
}
