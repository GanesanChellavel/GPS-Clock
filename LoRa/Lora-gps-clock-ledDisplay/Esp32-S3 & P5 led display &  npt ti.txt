#include <Arduino.h>
#include <ESP32-HUB75-MatrixPanel-I2S-DMA.h>
#include <WiFi.h>
#include "time.h"

// ---- Panel config ----
#define PANEL_WIDTH    64
#define PANEL_HEIGHT   32
#define PANELS_NUMBER  1

// ---- WiFi ----
const char* ssid     = "NOKIA-9A21";
const char* password = "59vtvyRpBq";

// ---- NTP (Singapore UTC+8) ----
const char* ntpServer          = "pool.ntp.org";
const long  gmtOffset_sec      = 8 * 3600;
const int   daylightOffset_sec = 0;

MatrixPanel_I2S_DMA *dma_display = nullptr;
uint16_t colorYellow, colorBlack;

// ---- WiFi + NTP Sync ----
void connectWiFiAndSync() {
  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi connected");

  configTime(gmtOffset_sec, daylightOffset_sec, ntpServer);

  struct tm timeinfo;
  Serial.print("Waiting for NTP sync");
  while (!getLocalTime(&timeinfo)) {
    Serial.print(".");
    delay(500);
  }
  Serial.println("\nTime synced!");

  WiFi.disconnect(true);
  WiFi.mode(WIFI_OFF);
}

// ---- Show HH:MM ----
void showTime() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) return;

  char timeStr[6];
  strftime(timeStr, sizeof(timeStr), "%H:%M", &timeinfo);

  dma_display->clearScreen();
  dma_display->setTextSize(2);
  dma_display->setTextColor(colorYellow);
  dma_display->setTextWrap(false);

  int x = (PANEL_WIDTH - (5 * 12)) / 2;
  int y = (PANEL_HEIGHT - 16) / 2;

  dma_display->setCursor(x, y);
  dma_display->print(timeStr);

  Serial.println(timeStr);
}

void setup() {
  Serial.begin(115200);

  // ---- ESP32-S3 specific GPIO mapping ----
  HUB75_I2S_CFG::i2s_pins _pins = {
    .r1 = 1,
    .g1 = 2,
    .b1 = 3,
    .r2 = 4,
    .g2 = 5,
    .b2 = 6,
    .a  = 7,
    .b  = 8,
    .c  = 9,
    .d  = 10,
    .e  = 11,   // Remove if 1/16 scan panel
    .lat = 13,
    .oe  = 14,
    .clk = 12
  };

  HUB75_I2S_CFG mxconfig(PANEL_WIDTH, PANEL_HEIGHT, PANELS_NUMBER, _pins);
  mxconfig.clkphase = false;   // Fix for S3 timing
  mxconfig.i2sspeed = HUB75_I2S_CFG::HZ_10M; // Lower speed for stability

  dma_display = new MatrixPanel_I2S_DMA(mxconfig);
  dma_display->begin();
  dma_display->setBrightness8(100);
  dma_display->clearScreen();

  colorYellow = dma_display->color565(255, 220, 0);
  colorBlack  = dma_display->color565(0, 0, 0);

  // Show "Wait..." while syncing
  dma_display->setTextSize(1);
  dma_display->setTextColor(colorYellow);
  dma_display->setCursor(8, 12);
  dma_display->print("Wait...");

  connectWiFiAndSync();
}

void loop() {
  showTime();
  delay(1000);
}
